// ============================================================ // QuantBR — API de alertas (Netlify Function) // O site chama esta função para criar, listar e remover alertas. // Ela conversa com o Supabase usando a chave secreta que fica // AQUI no servidor (nunca no navegador). // // Rotas (via método HTTP): // POST { email, ticker, condition, target } -> cria alerta // GET ?email=... -> lista alertas do e-mail // DELETE ?id=... -> remove alerta // ============================================================ export default async (req) => { const { SUPABASE_URL, SUPABASE_SERVICE_KEY } = process.env; if (!SUPABASE_URL || !SUPABASE_SERVICE_KEY) { return json({ error: 'Servidor sem configuração do Supabase.' }, 500); } const sb = (path, opts = {}) => fetch(`${SUPABASE_URL}/rest/v1/${path}`, { ...opts, headers: { apikey: SUPABASE_SERVICE_KEY, Authorization: `Bearer ${SUPABASE_SERVICE_KEY}`, 'Content-Type': 'application/json', ...(opts.headers || {}), }, }); const url = new URL(req.url); try { // ---- criar alerta ---- if (req.method === 'POST') { const body = await req.json(); const { email, ticker, condition, target } = body || {}; // validações simples if (!email || !/^[^@]+@[^@]+\.[^@]+$/.test(email)) return json({ error: 'E-mail inválido.' }, 400); if (!ticker) return json({ error: 'Ticker obrigatório.' }, 400); if (!['above', 'below'].includes(condition)) return json({ error: 'Condição inválida.' }, 400); if (isNaN(Number(target))) return json({ error: 'Preço-alvo inválido.' }, 400); const res = await sb('alerts', { method: 'POST', headers: { Prefer: 'return=representation' }, body: JSON.stringify({ email: String(email).toLowerCase().trim(), ticker: String(ticker).toUpperCase().trim(), condition, target: Number(target), triggered: false, }), }); if (!res.ok) return json({ error: await res.text() }, 500); return json({ ok: true, alert: (await res.json())[0] }); } // ---- listar alertas de um e-mail ---- if (req.method === 'GET') { const email = (url.searchParams.get('email') || '').toLowerCase().trim(); if (!email) return json({ error: 'Informe o e-mail.' }, 400); const res = await sb(`alerts?email=eq.${encodeURIComponent(email)}&order=created_at.desc&select=*`); if (!res.ok) return json({ error: await res.text() }, 500); return json({ ok: true, alerts: await res.json() }); } // ---- remover alerta ---- if (req.method === 'DELETE') { const id = url.searchParams.get('id'); if (!id) return json({ error: 'Informe o id.' }, 400); const res = await sb(`alerts?id=eq.${encodeURIComponent(id)}`, { method: 'DELETE', headers: { Prefer: 'return=minimal' }, }); if (!res.ok) return json({ error: await res.text() }, 500); return json({ ok: true }); } return json({ error: 'Método não suportado.' }, 405); } catch (e) { return json({ error: String(e) }, 500); } }; function json(obj, status = 200) { return new Response(JSON.stringify(obj), { status, headers: { 'Content-Type': 'application/json', 'Access-Control-Allow-Origin': '*' }, }); }
